## 源文件规格（逐字骨架，`packages-user/client-base/src/map/mapText.tsx`）

```tsx
import { MotaOffscreenCanvas2D } from '@motajs/render';
import { DefaultProps } from '@motajs/render-vue';
import { defineComponent } from 'vue';
import { SetupComponentOptions } from '@motajs/system';
import { IMapRenderer } from './types';
import { IMapTextArea, IMapTextRenderable } from './extension/types';

export interface IMapTextManager {
    /**
     * 申请指定图块坐标的文字管理对象，如果该点已被申请，则会使用已申请的对象
     * @param x 图块横坐标
     * @param y 图块纵坐标
     * @returns 申请的文字管理对象，如果申请的坐标不在地图上，则会返回 `null`
     */
    requireBlockArea(x: number, y: number): Readonly<IMapTextArea> | null;

    /**
     * 根据图块坐标获取已申请的文字管理对象，如果未申请则返回 `null`
     * @param x 图块横坐标
     * @param y 图块纵坐标
     */
    getBlockByLoc(x: number, y: number): Readonly<IMapTextArea> | null;

    /**
     * 根据索引获取已申请的文字管理对象，如果未申请则返回 `null`
     * @param index 管理对象索引
     */
    getBlockByIndex(index: number): Readonly<IMapTextArea> | null;

    /**
     * 获取指定文字管理对象中的所有可渲染对象
     * @param area 文字管理对象
     */
    getRenderables(area: Readonly<IMapTextArea>): Iterable<IMapTextRenderable>;

    /**
     * 判断是否需要更新
     */
    needUpdate(): boolean;

    /**
     * 清空所有文字内容
     */
    clear(): void;

    /**
     * 摧毁这个文字管理对象，释放相关资源
     */
    destroy(): void;
}

export interface IMapTextProps extends DefaultProps {
    /** 文字管理器，提供文字数据 */
    manager: IMapTextManager;
    /** 地图渲染器，提供尺寸与可见区域 */
    renderer: IMapRenderer;
}

const mapTextProps = {
    props: ['manager', 'renderer']
} satisfies SetupComponentOptions<IMapTextProps>;

/**
 * 地图文字组件，按照地图坐标系绘制文字管理器中的所有文字。
 * 参数为文字管理器与地图渲染器，前者提供文字数据，后者提供尺寸与可见区域。
 */
export const MapText = defineComponent<IMapTextProps>(props => {
    // TODO: 管理器内容与变换矩阵的更新尚未接线，当前依赖 custom 元素自身的缓存
    const render = (canvas: MotaOffscreenCanvas2D) => {
        const { manager, renderer } = props;
        const ctx = canvas.ctx;
        const block = renderer.vertex.block;
        const blockList = renderer.viewport.getRenderArea().blockList;
        ctx.lineWidth = 2;
        for (const chunk of blockList) {
            const cells = block.iterateElementsOfChunkIndex(chunk.index);
            for (const cell of cells) {
                const area = manager.getBlockByLoc(cell.x, cell.y);
                if (area) {
                    const baseX = area.mapX * renderer.cellWidth;
                    const baseY = area.mapY * renderer.cellHeight;
                    const renderables = manager.getRenderables(area);
                    for (const renderable of renderables) {
                        const x = baseX + (renderable.px ?? 0);
                        const y = baseY + (renderable.py ?? 0);
                        ctx.font = renderable.font.string();
                        ctx.textAlign = renderable.textAlign ?? 'left';
                        ctx.textBaseline = renderable.textBaseline ?? 'top';
                        if (renderable.stroke) {
                            ctx.strokeStyle = renderable.strokeStyle ?? 'black';
                            ctx.strokeText(renderable.text, x, y);
                        }
                        if (renderable.fill) {
                            ctx.fillStyle = renderable.fillStyle ?? 'white';
                            ctx.fillText(renderable.text, x, y);
                        }
                    }
                }
            }
        }
    };

    return () => {
        const { renderWidth, renderHeight } = props.renderer;
        return (
            <custom
                selfTransform
                loc={[0, 0, renderWidth, renderHeight]}
                render={render}
            />
        );
    };
}, mapTextProps);
```

要点（写代码时逐条自检）：

- **不写任何矩阵代码**：没有 `ctx.transform(` / `ctx.setTransform(` / `ctx.translate(` / `ctx.rotate(` / `ctx.scale(`，也没有 `scale(1, -1)` / `scale(-1, 1)`——变换由 `selfTransform` 在元素内部应用。
- **正向分支**：用 `if (area) { ... }` 包住本格绘制（等价于跳过空格子，但不使用 `if (!area)` 早退）。
- **`selfTransform` 写成裸布尔属性**（`react/jsx-boolean-value` 规则要求）；盒子只给 `loc`，不给 `transform`。
- `<custom>` 默认走缓存路径，画布在渲染前由渲染系统清空，因此组件内**不写** `clearRect`；**不传** `noanti`（文字需要抗锯齿）。
- `// TODO:` 文案只说明「更新尚未接线、当前依赖缓存」，不得暗示已实现。
- 组件顶层无副作用；`render` 在回调内才读取 `props`。

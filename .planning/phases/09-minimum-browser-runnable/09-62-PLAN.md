## 仅注释差分闸门设计（`git diff` 的等价实现）

**为什么不直接用 `git diff -U0 -- <3 files>`：** `probe.ts` / `opusSession.ts` 是 09-61 新建但**未提交**的未跟踪文件，`git diff` 对它们输出为空；`context.ts` 已带有 09-61 的未提交改动，`git diff` 相对 `HEAD` 的结果混入了 09-61 的逻辑改动，无法单独表达本计划的「只加注释」。使用 `git add -N` 会把未跟踪文件写进索引，属于 git 写命令，被本计划禁止。

**等价实现：** Task 1 第 0 步把三个目标文件的**改前字节**复制到临时目录（`gsd-09-62-before-probe.ts` / `gsd-09-62-before-opusSession.ts` / `gsd-09-62-before-context.ts`），Task 1 / Task 2 用 `gsd-09-62-gate-delta.cjs` 断言：

1. 对每个文件做「改前行序列是否为改后行序列的子序列」检查——存在删除行或修改行都会导致子序列匹配失败；
2. 改后多出来的行（inserted）必须全部是注释行：`trim()` 后以 `/**`、`*`、`*/` 或 `//` 开头；
3. 三个文件 inserted 中 `/**` 开头的行总数必须**恰为 15**（5 个单行字段注释 + 10 个多行方法/函数注释各一个开头），多一个或少一个都失败。

**允许的偏差：** 本计划不使用 `prettier --write`，只跑 `--check`；若 `--check` 失败且仅由注释插入导致，执行器**停下上报**，不得为了让 `--check` 过而重排代码（重排会被本闸门判为「修改代码行」）。

### `gsd-09-62-gate-delta.cjs`（只加注释 + 恰好 15 个注释块）

```js
const fs = require('fs');
const path = require('path');

const temp = 'C:/Users/13194/AppData/Local/Temp/opencode';
const pairs = [
    { name: 'probe.ts', before: path.join(temp, 'gsd-09-62-before-probe.ts'), after: 'packages/audio/src/probe.ts' },
    { name: 'opusSession.ts', before: path.join(temp, 'gsd-09-62-before-opusSession.ts'), after: 'packages/audio/src/opusSession.ts' },
    { name: 'context.ts', before: path.join(temp, 'gsd-09-62-before-context.ts'), after: 'packages/audio/src/context.ts' }
];

function lines(file) {
    return fs.readFileSync(file, 'utf8').replace(/\r\n/g, '\n').split('\n');
}

const failures = [];
let insertedTotal = 0;
let openedBlocks = 0;

for (const pair of pairs) {
    if (!fs.existsSync(pair.before)) {
        failures.push(pair.name + ' 缺少改前副本 ' + pair.before);
        continue;
    }
    const before = lines(pair.before);
    const after = lines(pair.after);
    let cursor = 0;
    const inserted = [];
    for (const line of after) {
        if (cursor < before.length && line === before[cursor]) {
            cursor++;
        } else {
            inserted.push(line);
        }
    }
    if (cursor !== before.length) {
        failures.push(pair.name + ' 改前内容不是改后内容的子序列（存在删除或修改的行）');
        continue;
    }
    const bad = inserted.filter(line => {
        const t = line.trim();
        return !(t.startsWith('/**') || t.startsWith('*/') || t.startsWith('*') || t.startsWith('//'));
    });
    if (bad.length) {
        failures.push(pair.name + ' 存在非注释新增行：' + JSON.stringify(bad.slice(0, 3)));
    }
    const opened = inserted.filter(line => line.trim().startsWith('/**')).length;
    openedBlocks += opened;
    insertedTotal += inserted.length;
    console.log(pair.name + ': inserted=' + inserted.length + ', opened JSDoc blocks=' + opened);
}

if (openedBlocks !== 15) {
    failures.push('新增 /** 注释块共 ' + openedBlocks + ' 个，期望恰 15 个');
}

if (failures.length) {
    console.error('delta gate failed:');
    for (const failure of failures) console.error('  - ' + failure);
    process.exit(1);
}
console.log('OK: only comment lines inserted (' + insertedTotal + ' lines, 15 blocks), no removed or modified code lines');
```

### 2. `packages/audio/src/probe.ts`（全文替换）

```ts
import {
    BlobSource,
    BufferSource,
    Input,
    OGG,
    UrlSource,
    WEBM
} from 'mediabunny';
import { logger } from '@motajs/common';
import {
    AudioCodec,
    AudioContainer,
    AudioInputSource,
    IAudioFormatInfo
} from './types';

// 只声明本 tracer 支持的容器格式，避免为其它格式付解析器体积
const mediaFormats = [OGG, WEBM];

/**
 * 把统一的音频输入类型转换为 mediabunny 支持的数据源，不支持的输入返回 null
 * @param source 音频输入数据
 */
function createMediaSource(
    source: AudioInputSource
): BufferSource | BlobSource | UrlSource | null {
    if (source instanceof ArrayBuffer) return new BufferSource(source);
    if (source instanceof Uint8Array) return new BufferSource(source);
    if (source instanceof Blob) return new BlobSource(source);
    if (typeof source === 'string') return new UrlSource(source);
    logger.warn(217, source.constructor.name);
    return null;
}

/**
 * 用 Ogg / WebM 精选格式表创建一个 mediabunny 输入
 * @param source 音频输入数据
 */
export function createMediaInput(source: AudioInputSource): Input | null {
    const mediaSource = createMediaSource(source);
    if (!mediaSource) return null;
    return new Input({ formats: mediaFormats, source: mediaSource });
}

/**
 * 把 mediabunny 的编码字符串映射为本包的音频编码枚举
 * @param codec mediabunny 返回的编码字符串
 */
function toAudioCodec(codec: string | null): AudioCodec {
    switch (codec) {
        case 'opus':
            return AudioCodec.Opus;
        case 'vorbis':
            return AudioCodec.Vorbis;
        case 'aac':
            return AudioCodec.Aac;
        case 'mp3':
            return AudioCodec.Mp3;
        case 'flac':
            return AudioCodec.Flac;
        case 'pcm-s16':
        case 'pcm-s24':
        case 'pcm-s32':
        case 'pcm-f32':
            return AudioCodec.Pcm;
        default:
            return AudioCodec.Unknown;
    }
}

/**
 * 探测音频数据的容器、编码、采样率、声道数与总时长
 * @param source 音频输入数据
 */
export async function probeAudioSource(
    source: AudioInputSource
): Promise<IAudioFormatInfo | null> {
    const input = createMediaInput(source);
    if (!input) return null;
    const readable = await input.canRead();
    if (!readable) {
        input.dispose();
        return null;
    }
    const format = await input.getFormat();
    let container = AudioContainer.Unknown;
    if (format === OGG) container = AudioContainer.Ogg;
    else if (format === WEBM) container = AudioContainer.WebM;
    if (container === AudioContainer.Unknown) {
        // 已注册格式表只有 Ogg / WebM，走到这里说明探测结果异常
        input.dispose();
        logger.error(97);
        return null;
    }
    const track = await input.getPrimaryAudioTrack();
    if (!track) {
        input.dispose();
        logger.error(97);
        return null;
    }
    const codec = toAudioCodec(await track.getCodec());
    const sampleRate = await track.getSampleRate();
    const channels = await track.getNumberOfChannels();
    const duration = await input.computeDuration();
    input.dispose();
    return { container, codec, sampleRate, channels, duration };
```

```ts
 * @param description 解码器描述数据，即 OpusHead
 * @returns 解析得到的识别头，缺少或非法时为 null

export function parseOpusIdentificationHeader(
    description: Uint8Array | null
): IOpusIdentificationHeader | null {
    if (!description || description.length < 19) return null;
    const magic = new TextDecoder().decode(description.slice(0, 8));
    if (magic !== 'OpusHead') return null;
    const channelCount = description[9];
    if (channelCount < 1) return null;
    const view = new DataView(
        description.buffer,
        description.byteOffset,
        description.byteLength
    );
    const preSkip = view.getUint16(10, true);
    const mappingFamily = description[18];
    if (mappingFamily !== 0 && description.length < 21 + channelCount) {
        return null;
    }
    if (mappingFamily === 0) {
        return {
            preSkip,
            channels: channelCount,
            streamCount: 1,
            coupledStreamCount: channelCount - 1,
            channelMappingTable: channelCount === 2 ? [0, 1] : [0]
        };
    }
    const streamCount = description[19];
    const coupledStreamCount = description[20];
    const channelMappingTable: number[] = [];
    for (let i = 0; i < channelCount; i++) {
        channelMappingTable.push(description[21 + i]);
    }
    return {
        preSkip,
        channels: channelCount,
        streamCount,
        coupledStreamCount,
        channelMappingTable
    };
}

    header: IOpusIdentificationHeader
): Promise<IOpusDecoderInstance> {
    let pending = opusModulePromise;
    if (!pending) {
        pending = import('opus-decoder');
        opusModulePromise = pending;
    }
    const opus = await pending.catch(error => {
        // 加载失败时清空缓存，保证后续调用可以重试
        opusModulePromise = null;
        throw error;
    });

        sampleRate: 48000,
        channels: header.channels,
        preSkip: header.preSkip,
        streamCount: header.streamCount,
        coupledStreamCount: header.coupledStreamCount,
        channelMappingTable: header.channelMappingTable
    };

export class OpusDecodeSession implements IAudioDecodeSession {
    readonly info: IAudioFormatInfo;
    readonly chunks: AsyncIterable<IAudioDecodedChunk>;

    /** 会话持有的 mediabunny 输入，关闭会话时释放 */
    private readonly input: Input;
    /** 按包顺序读取编码数据的包读取器 */
    private readonly sink: EncodedPacketSink;
    /** 当前会话使用的 Opus 解码器实例 */
    private readonly decoder: IOpusDecoderInstance;
    /** 会话是否已关闭，用于中止异步迭代 */
    private closed: boolean = false;

    constructor(
        input: Input,
        sink: EncodedPacketSink,
        info: IAudioFormatInfo,
        decoder: IOpusDecoderInstance
    ) {
        this.input = input;
        this.sink = sink;
        this.info = info;
        this.decoder = decoder;
        this.chunks = this.generateChunks();
    }

    /**
     * 解码单个 Opus 包，空包返回 null
     * @param packet 待解码的编码包
     */
    private async decodePacket(
        packet: EncodedPacket
    ): Promise<IAudioDecodedChunk | null> {
        const decoded = await this.decoder.decodeFrame(packet.data);
        // 零采样但带解码错误时需要保留错误，交给上层记录失败
        const isEmpty = decoded.samplesDecoded === 0;
        if (isEmpty && decoded.errors.length === 0) return null;
        return {
            channelData: decoded.channelData.map(
                channel => new Float32Array(channel)
            ),
            samplesDecoded: decoded.samplesDecoded,
            sampleRate: decoded.sampleRate,
            timestamp: packet.timestamp,
            duration: packet.duration,
            errors: decoded.errors
        };
    }

    /**
     * 顺序读取并解码全部 Opus 包，会话关闭后提前结束
     */
    private async *generateChunks(): AsyncGenerator<IAudioDecodedChunk> {
        for await (const packet of this.sink.packets()) {
            if (this.closed) return;
            const chunk = await this.decodePacket(packet);
            if (chunk) yield chunk;
        }
    }

    close(): void {
        if (this.closed) return;
        this.closed = true;
        this.input.dispose();
        this.decoder.free();
    }
```

```ts
import {
    BlobSource,
    BufferSource,
    Input,
    OGG,
    UrlSource,
    WEBM
} from 'mediabunny';
import { AudioCodec, AudioContainer, AudioInputSource, IAudioFormatInfo } from './types';

// 只声明本 tracer 支持的容器格式，避免为其它格式付解析器体积
const mediaFormats = [OGG, WEBM];

function createMediaSource(
    source: AudioInputSource
): BufferSource | BlobSource | UrlSource | null {
    if (source instanceof ArrayBuffer) return new BufferSource(source);
    if (source instanceof Uint8Array) return new BufferSource(source);
    if (source instanceof Blob) return new BlobSource(source);
    if (typeof source === 'string') return new UrlSource(source);
    return null;
}

/**
 * 用 Ogg / WebM 精选格式表创建一个 mediabunny 输入
 * @param source 音频输入数据
 */
export function createMediaInput(source: AudioInputSource): Input | null {
    const mediaSource = createMediaSource(source);
    if (!mediaSource) return null;
    return new Input({ formats: mediaFormats, source: mediaSource });
}

function toAudioCodec(codec: string | null): AudioCodec {
    switch (codec) {
        case 'opus':
            return AudioCodec.Opus;
        case 'vorbis':
            return AudioCodec.Vorbis;
        case 'aac':
            return AudioCodec.Aac;
        case 'mp3':
            return AudioCodec.Mp3;
        case 'flac':
            return AudioCodec.Flac;
        case 'pcm-s16':
        case 'pcm-s24':
        case 'pcm-s32':
        case 'pcm-f32':
            return AudioCodec.Pcm;
        default:
            return AudioCodec.Unknown;
    }
}

/**
 * 探测音频数据的容器、编码、采样率、声道数与总时长
 * @param source 音频输入数据
 */
export async function probeAudioSource(
    source: AudioInputSource
): Promise<IAudioFormatInfo | null> {
    const input = createMediaInput(source);
    if (!input) return null;
    const readable = await input.canRead();
    if (!readable) {
        input.dispose();
        return null;
    }
    const track = await input.getPrimaryAudioTrack();
    if (!track) {
        input.dispose();
        return null;
    }
    const format = await input.getFormat();
    let container = AudioContainer.Unknown;
    if (format === OGG) container = AudioContainer.Ogg;
    else if (format === WEBM) container = AudioContainer.WebM;
    if (container === AudioContainer.Unknown) {
        input.dispose();
        return null;
    }
    const codec = toAudioCodec(await track.getCodec());
    const sampleRate = await track.getSampleRate();
    const channels = await track.getNumberOfChannels();
    const duration = await input.computeDuration();
    input.dispose();
    return { container, codec, sampleRate, channels, duration };
}
```

**说明：** 只接受 `ArrayBuffer` / `Uint8Array` / `Blob` / `string`；`ReadableStream` 在 tracer 里不支持（返回 `null`，探测随之为 `null`），SUMMARY 记录该子集。`computeDuration()` 对整文件是常量级扫描，tracer 接受。

```ts
import { EncodedPacket, EncodedPacketSink, Input } from 'mediabunny';
import {
    AudioInputSource,
    IAudioDecodeError,
    IAudioDecodedChunk,
    IAudioDecodeSession,
    IAudioFormatInfo
} from './types';
import { createMediaInput } from './probe';

interface IOpusIdentificationHeader {
    readonly preSkip: number;
    readonly channels: number;
    readonly streamCount: number;
    readonly coupledStreamCount: number;
    readonly channelMappingTable: number[];
}

interface IOpusDecodedAudio {
    readonly channelData: Float32Array[];
    readonly samplesDecoded: number;
    readonly sampleRate: number;
    readonly errors: IAudioDecodeError[];
}

interface IOpusDecoderOptions {
    readonly sampleRate: 48000;
    readonly channels: number;
    readonly preSkip: number;
    readonly streamCount: number;
    readonly coupledStreamCount: number;
    readonly channelMappingTable: number[];
}

interface IOpusDecoderInstance {
    readonly ready: Promise<void>;

    decodeFrame(packet: Uint8Array): IOpusDecodedAudio | Promise<IOpusDecodedAudio>;

    free(): void | Promise<void>;
}

interface IOpusDecoderModule {
    readonly OpusDecoder: new (options: IOpusDecoderOptions) => IOpusDecoderInstance;
    readonly OpusDecoderWebWorker: new (
        options: IOpusDecoderOptions
    ) => IOpusDecoderInstance;
}

/**
 * 解析 Opus 识别头，缺少或非法时按声道数回退到默认映射
 * @param description 解码器描述数据，即 OpusHead
 * @param channels 音频声道数
 */
export function parseOpusIdentificationHeader(
    description: Uint8Array | null,
    channels: number
): IOpusIdentificationHeader {
    const fallback: IOpusIdentificationHeader = {
        preSkip: 0,
        channels,
        streamCount: 1,
        coupledStreamCount: channels === 2 ? 1 : 0,
        channelMappingTable: channels === 2 ? [0, 1] : [0]
    };
    if (!description || description.length < 19) return fallback;
    const magic = new TextDecoder().decode(description.slice(0, 8));
    if (magic !== 'OpusHead') return fallback;
    const channelCount = description[9];
    if (channelCount < 1) return fallback;
    const view = new DataView(
        description.buffer,
        description.byteOffset,
        description.byteLength
    );
    const preSkip = view.getUint16(10, true);
    const mappingFamily = description[18];
    if (mappingFamily !== 0 && description.length < 21 + channelCount) {
        return fallback;
    }
    if (mappingFamily === 0) {
        return {
            preSkip,
            channels: channelCount,
            streamCount: 1,
            coupledStreamCount: channelCount - 1,
            channelMappingTable: channelCount === 2 ? [0, 1] : [0]
        };
    }
    const streamCount = description[19];
    const coupledStreamCount = description[20];
    const channelMappingTable: number[] = [];
    for (let i = 0; i < channelCount; i++) {
        channelMappingTable.push(description[21 + i]);
    }
    return {
        preSkip,
        channels: channelCount,
        streamCount,
        coupledStreamCount,
        channelMappingTable
    };
}

export class OpusDecodeSession implements IAudioDecodeSession {
    readonly info: IAudioFormatInfo;
    readonly chunks: AsyncIterable<IAudioDecodedChunk>;

    private static opusModule: IOpusDecoderModule | null = null;

    private readonly input: Input;
    private readonly sink: EncodedPacketSink;
    private readonly decoder: IOpusDecoderInstance;
    private closed: boolean = false;

    private static async loadDecoder(
        header: IOpusIdentificationHeader
    ): Promise<IOpusDecoderInstance> {
        let opus = OpusDecodeSession.opusModule;
        if (!opus) {
            opus = await import('opus-decoder');
            OpusDecodeSession.opusModule = opus;
        }
        const options: IOpusDecoderOptions = {
            sampleRate: 48000,
            channels: header.channels,
            preSkip: header.preSkip,
            streamCount: header.streamCount,
            coupledStreamCount: header.coupledStreamCount,
            channelMappingTable: header.channelMappingTable
        };
        const decoder: IOpusDecoderInstance =
            typeof Worker === 'undefined'
                ? new opus.OpusDecoder(options)
                : new opus.OpusDecoderWebWorker(options);
        await decoder.ready;
        return decoder;
    }

    /**
     * 为 Ogg / WebM Opus 创建整文件解码会话
     * @param source 音频输入数据
     * @param info 探测得到的格式信息
     */
    static async create(
        source: AudioInputSource,
        info: IAudioFormatInfo
    ): Promise<OpusDecodeSession | null> {
        const input = createMediaInput(source);
        if (!input) return null;
        const readable = await input.canRead();
        if (!readable) {
            input.dispose();
            return null;
        }
        const track = await input.getPrimaryAudioTrack();
        if (!track) {
            input.dispose();
            return null;
        }
        const config = await track.getDecoderConfig();
        let description: Uint8Array | null = null;
        if (config && config.description instanceof Uint8Array) {
            description = config.description;
        }
        const header = parseOpusIdentificationHeader(description, info.channels);
        const decoder = await OpusDecodeSession.loadDecoder(header);
        const sink = new EncodedPacketSink(track);
        return new OpusDecodeSession(input, sink, info, decoder);
    }

    private async decodePacket(
        packet: EncodedPacket
    ): Promise<IAudioDecodedChunk | null> {
        const decoded = await this.decoder.decodeFrame(packet.data);
        if (decoded.samplesDecoded === 0) return null;
        return {
            channelData: decoded.channelData.map(
                channel => new Float32Array(channel)
            ),
            samplesDecoded: decoded.samplesDecoded,
            sampleRate: decoded.sampleRate,
            timestamp: packet.timestamp,
            duration: packet.duration,
            errors: decoded.errors
        };
    }

    private async *generateChunks(): AsyncGenerator<IAudioDecodedChunk> {
        for await (const packet of this.sink.packets()) {
            if (this.closed) return;
            const chunk = await this.decodePacket(packet);
            if (chunk) yield chunk;
        }
    }

    constructor(
        input: Input,
        sink: EncodedPacketSink,
        info: IAudioFormatInfo,
        decoder: IOpusDecoderInstance
    ) {
        this.input = input;
        this.sink = sink;
        this.info = info;
        this.decoder = decoder;
        this.chunks = this.generateChunks();
    }

    close(): void {
        if (this.closed) return;
        this.closed = true;
        this.input.dispose();
        this.decoder.free();
    }
}
```

**说明：** `decodeFrame` 在主线程实现里同步返回、在 WebWorker 实现里返回 Promise，故实例接口声明为联合后统一 `await`。`channelData` 逐声道复制成新的 `Float32Array`，既满足 `Float32Array<ArrayBuffer>` 的接口形状，也避免与 `opus-decoder` 的复用缓冲共享内存。**调用方必须 `close()`**（`context.decodeSession` 会调用）。探测（`probeAudioSource`）与会话各建一次 `Input`（多一次头部解析），tracer 接受该成本，合并留给 09-63。

    private async decodeNative(data: Uint8Array): Promise<AudioBuffer | null> {
        const copy = data.slice();
        try {
            return await this.ac.decodeAudioData(copy.buffer);
        } catch {
            return null;
        }

    private convertAudioBuffer(buffer: AudioBuffer): IAudioDecodeData {
        const channelData: Float32Array<ArrayBuffer>[] = [];
        for (let i = 0; i < buffer.numberOfChannels; i++) {
            channelData.push(buffer.getChannelData(i));
        }
        return {
            channelData,
            samplesDecoded: buffer.length,
            sampleRate: buffer.sampleRate,
            errors: []
        };
    }

    private mergeChannelData(
        chunks: Float32Array<ArrayBuffer>[]
    ): Float32Array<ArrayBuffer> {
        const total = chunks.reduce((prev, curr) => prev + curr.length, 0);
        const merged = new Float32Array(total);
        let offset = 0;
        for (const chunk of chunks) {
            merged.set(chunk, offset);
            offset += chunk.length;
        }
        return merged;
    }

    private async decodeSession(
        session: IAudioDecodeSession
    ): Promise<IAudioDecodeData | null> {
        const channelChunks: Float32Array<ArrayBuffer>[][] = [];
        const errors: IAudioDecodeError[] = [];
        let samplesDecoded = 0;
        let sampleRate = 0;
        for await (const chunk of session.chunks) {
            for (let i = 0; i < chunk.channelData.length; i++) {
                if (!channelChunks[i]) channelChunks[i] = [];
                channelChunks[i].push(chunk.channelData[i]);
            }
            samplesDecoded += chunk.samplesDecoded;
            sampleRate = chunk.sampleRate;
            errors.push(...chunk.errors);
        }
        session.close();
        if (samplesDecoded === 0) return null;
        const channelData = channelChunks.map(chunks =>
            this.mergeChannelData(chunks)
        );
        return {
            channelData,
            samplesDecoded: channelData[0].length,
            sampleRate,
            errors
        };
    }

    private async decodeAudioLegacy(
        data: Uint8Array
    ): Promise<IAudioDecodeData | null> {
        const type = this.getAudioTypeFromData(data);
        if (type === AudioType.Unknown) {
            logger.error(25, this.getErrorHeaderInfo(data));
            return null;
        }
        const decoder = this.createDecoder(type);
        if (!decoder) {
            logger.error(25, this.getErrorHeaderInfo(data));
            return null;
        }
        await decoder.create();
        const decoded = await decoder.decodeAll(data);
        await decoder.destroy();
        return decoded;
    }
